## 🛠️ 9. คู่มือสคริปต์สำหรับการ Push ข้อมูลขึ้น GitHub (Deployment Guide)

เมื่อไม่มีคำสั่ง `git` ในเครื่อง ให้ใช้ PowerShell Script สั่งงานผ่าน GitHub REST API ดังนี้:

```powershell
# รันในโฟลเดอร์ g:\My Drive\ร้าน
$token = '<YOUR_GITHUB_PERSONAL_ACCESS_TOKEN>'
$repo = 'dewsun8-18/dew'
$headers = @{
    'Authorization' = "Bearer $token"
    'Accept' = 'application/vnd.github.v3+json'
    'User-Agent' = 'PowerShell-Script'
}

$allFiles = Get-ChildItem -Path . -File | Where-Object {
    $_.Extension -in @('.csv', '.html', '.md', '.xlsx') -or $_.Name -in @('dashboard.html', 'index.html', 'README.md')
}

foreach ($file in $allFiles) {
    $encodedName = [System.Uri]::EscapeDataString($file.Name)
    $url = "https://api.github.com/repos/$repo/contents/$encodedName"

    $sha = $null
    try {
        $resp = Invoke-RestMethod -Uri $url -Headers $headers -Method Get
        $sha = $resp.sha
    } catch {}

    $bytes = [System.IO.File]::ReadAllBytes($file.FullName)
    $base64Content = [System.Convert]::ToBase64String($bytes)

    $body = @{
        message = "Update $($file.Name)"
        content = $base64Content
        branch = 'main'
    }
    if ($sha) { $body['sha'] = $sha }

    $jsonBody = $body | ConvertTo-Json
    $putResp = Invoke-RestMethod -Uri $url -Headers $headers -Method Put -Body ([System.Text.Encoding]::UTF8.GetBytes($jsonBody)) -ContentType 'application/json; charset=utf-8'
    Write-Host "Successfully pushed: $($file.Name)"
}
```
